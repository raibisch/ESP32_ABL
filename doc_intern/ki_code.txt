#include <WiFi.h>
#include <ESPAsyncWebServer.h>
#include <ArduinoJson.h>

AsyncWebServer server(80);

// Verknüpfung mit deiner bestehenden ABL-Steuerung
void setWallboxPowerLimit(float kw) {
  // Dein Code: Berechne Ampere aus kW (z.B. 4.2 kW / 3 Phasen = ca. 6A)
  // Und sende es an das ABL-Protokoll
  Serial.printf("[§14a] Neues Limit gesetzt: %.2f kW\n", kw);
}

void setupAsyncAPI() {
  
  // POST-Endpunkt für die Leistungsbegrenzung
  server.on("/api/v1/charge_limit", HTTP_POST, [](AsyncWebServerRequest *request){}, 
    NULL, // Kein File-Upload
    [](AsyncWebServerRequest *request, uint8_t *data, size_t len, size_t index, size_t total) {
      
      // JSON-Dokument vorbereiten
      JsonDocument doc;
      DeserializationError error = deserializeJson(doc, (const char*)data, len);

      // Fehlerhaftes JSON abfangen
      if (error) {
        request->send(400, "application/json", "{\"error\":\"Invalid JSON\"}");
        return;
      }

      // Prüfen, ob der Schlüssel existiert
      if (doc.containsKey("max_power_kw")) {
        float limit_kw = doc["max_power_kw"];

        // Plausibilitätsprüfung (0.0 für Stop, 4.2 für §14a, bis max. 11/22 kW)
        if (limit_kw >= 0.0 && limit_kw <= 22.0) {
          
          // Hier wird deine bestehende Steuerfunktion aufgerufen
          setWallboxPowerLimit(limit_kw);

          // JSON-Antwort erstellen
          JsonDocument responseDoc;
          responseDoc["status"] = "success";
          responseDoc["limited_to_kw"] = limit_kw;

          String responseString;
          serializeJson(responseDoc, responseString);
          
          request->send(200, "application/json", responseString);
          return;
        }
      }

      request->send(400, "application/json", "{\"error\":\"Missing or invalid max_power_kw\"}");
    }
  );

  // Server starten
  server.begin();
}

void setup() {
  Serial.begin(115200);
  // WiFi-Verbindung hier initialisieren...
  
  setupAsyncAPI();
}

void loop() {
  // Der AsyncWebServer benötigt KEINEN Aufruf im loop().
  // Er läuft vollautomatisch im Hintergrund auf dem ESP32-Netzwerk-Stack!
}
